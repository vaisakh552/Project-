    function normalise(s) {
      return s.toLowerCase().replace(/@/g, 'a').replace(/0/g, 'o').replace(/[1!|]/g, 'i').replace(/3/g, 'e').replace(/[$5]/g, 's').replace(/7/g, 't');

    function hasSequence(pw) {
      var s = pw.toLowerCase();
      for (var i = 0; i < s.length - 2; i++) {
        var a = s.charCodeAt(i), b = s.charCodeAt(i + 1), c = s.charCodeAt(i + 2);
        if ((b - a === 1 && c - b === 1) || (a - b === 1 && b - c === 1)) return true;

    }
    function plural(v, unit) { var r = Math.max(1, Math.round(v)); return r + ' ' + unit + (r === 1 ? '' : 's'); }
    function formatTime(sec) {
      var MIN = 60, H = 3600, D = 86400, MO = 2629800, Y = 31557600;
      if (!isFinite(sec)) return 'Billions of years+';
      if (sec < 1) return 'Instantly';
      if (sec < MIN) return plural(sec, 'second');
      if (sec < H) return plural(sec / MIN, 'minute');
      if (sec < D) return plural(sec / H, 'hour');
      if (sec < MO) return plural(sec / D, 'day');
      if (sec < Y) return plural(sec / MO, 'month');
      var y = sec / Y;
      if (y < 1000) return plural(y, 'year');
      if (y < 1e6) return Math.round(y / 1000) + ' thousand years';
      if (y < 1e9) return Math.round(y / 1e6) + ' million years';
      return 'Billions of years+';
    }

    function analyse(pw) {
      var checks = {
        length: pw.length >= 12,
        upper: /[A-Z]/.test(pw),
        lower: /[a-z]/.test(pw),
        number: /\d/.test(pw),
        symbol: /[^A-Za-z0-9]/.test(pw),
        common: pw.length > 0 && !isCommon(pw)
      };
      var pool = (checks.lower ? 26 : 0) + (checks.upper ? 26 : 0) + (checks.number ? 10 : 0) + (checks.symbol ? 33 : 0);
      var entropy = pw.length ? pw.length * Math.log2(pool || 1) : 0;
      var repeated = /(.)\1{2,}/.test(pw), sequence = hasSequence(pw);
      if (repeated) entropy *= 0.8;
      if (sequence) entropy *= 0.85;
      if (!checks.common && pw.length) entropy = Math.min(entropy, 12);

      var score;
      if (entropy < 28) score = 0;
      else if (entropy < 40) score = 1;
      else if (entropy < 60) score = 2;
      else if (entropy < 80) score = 3;
      else score = 4;
      if (pw.length < 8) score = Math.min(score, 1);
      if (!checks.common) score = 0;

      var seconds = Math.pow(2, entropy - 1) / 1e10;
      var tips = [];
      if (!checks.length) tips.push('Make it at least 12 characters long (16+ is even better).');
      if (!checks.upper) tips.push('Add an uppercase letter.');
      if (!checks.lower) tips.push('Add a lowercase letter.');
      if (!checks.number) tips.push('Include a number.');
      if (!checks.symbol) tips.push('Include a symbol such as ! # % or -.');
      if (!checks.common) tips.push('Avoid common words and patterns like “password”, “qwerty” or “123456”.');
      if (repeated) tips.push('Avoid repeated characters (aaa, 111).');
      if (sequence) tips.push('Avoid sequences such as abc, 123 or qwerty.');

      return { checks: checks, entropy: entropy, score: score, seconds: seconds, tips: tips };
    }

      toast('Generated password loaded into the checker.');
    });

    function copyText(text) {
      if (navigator.clipboard && window.isSecureContext) {
        return navigator.clipboard.writeText(text);
      }
      return new Promise(function (resolve, reject) {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        try { document.execCommand('copy') ? resolve() : reject(); } catch (e) { reject(e); }
        document.body.removeChild(ta);
      });
    }
    $('#genCopy').addEventListener('click', function () {
      copyText($('#genOut').textContent).then(function () { toast('Password copied to clipboard.'); }, function () { toast('Copy failed — please select and copy manually.'); });
    });
    generatePassword();

    /* =========================================================
       CYBERSECURITY AWARENESS QUIZ
       ========================================================= */
    var quizEl = $('#quizBody');
    var qs = [], cur = 0, score = 0, answers = [];
    var RING = 2 * Math.PI * 70;

    function renderIntro() {
      var best = store('cybersafe_best');
      quizEl.innerHTML =
        '<div class="quiz-intro">' +
          '<div class="quiz-badge">' + icon('shield-check') + '</div>' +
          '<h3>Cybersecurity Awareness Quiz</h3>' +
          '<p>Can you spot a scam, choose a safe password and protect your accounts? Let’s find out.</p>' +
          '<div class="quiz-meta">' +
            '<span>' + icon('clipboard') + QUESTIONS.length + ' questions</span>' +
            '<span>' + icon('clock') + 'About 3 minutes</span>' +
            '<span>' + icon('zap') + 'Instant feedback</span>' +
          '</div>' +
          '<button class="btn btn-primary" id="startQuiz">Start the quiz ' + icon('arrow-right') + '</button>' +
          (best !== null ? '<p class="best">Your best score: ' + best + '%</p>' : '') +
        '</div>';
      $('#startQuiz').addEventListener('click', startQuiz);
    }

    function startQuiz() {
      qs = shuffle(QUESTIONS).map(function (q) {
        return {
          q: q.q, explain: q.explain,
          opts: shuffle(q.options.map(function (t, i) { return { t: t, correct: i === q.answer }; }))
        };
      });
      cur = 0; score = 0; answers = [];
      renderQuestion();
    }

    function renderQuestion() {
      var q = qs[cur], last = cur === qs.length - 1;
      quizEl.innerHTML =
        '<div class="quiz-top"><span>' + icon('clipboard') + 'Question ' + (cur + 1) + ' of ' + qs.length + '</span><span id="quizScore">' + icon('award') + 'Score: ' + score + '</span></div>' +
        '<div class="progress"><i id="quizProgress" style="width:' + (cur / qs.length * 100) + '%"></i></div>' +
        '<h3 class="q-text">' + q.q + '</h3>' +
        '<div class="options">' + q.opts.map(function (o, i) {
          return '<button class="opt" data-i="' + i + '" style="animation-delay:' + (i * 0.05) + 's"><span class="letter">' + 'ABCD'.charAt(i) + '</span><span>' + o.t + '</span></button>';
        }).join('') + '</div>' +
        '<div class="explain" id="explain"></div>' +
        '<div class="quiz-foot">' +
          '<button class="btn btn-ghost btn-sm" id="restartQuiz">' + icon('refresh') + 'Restart</button>' +
          '<button class="btn btn-primary" id="nextQ" disabled>' + (last ? 'See my results' : 'Next question') + icon('arrow-right') + '</button>' +
        '</div>';
      $$('.opt', quizEl).forEach(function (b) { b.addEventListener('click', function () { choose(+b.dataset.i); }); });
      $('#nextQ').addEventListener('click', function () {
        if (cur < qs.length - 1) { cur++; renderQuestion(); } else { renderResult(); }
      });
      $('#restartQuiz').addEventListener('click', startQuiz);
    }

    function choose(i) {
      var q = qs[cur], btns = $$('.opt', quizEl), ok = q.opts[i].correct;
      btns.forEach(function (b, j) {
        b.disabled = true;
        if (q.opts[j].correct) b.classList.add('correct');
        else if (j !== i) b.classList.add('dim');
      });
      if (ok) score++; else btns[i].classList.add('wrong');
      answers.push({ q: q.q, chosen: q.opts[i].t, correct: q.opts.filter(function (o) { return o.correct; })[0].t, ok: ok });

      var ex = $('#explain');
      ex.innerHTML = '<strong>' + icon(ok ? 'check' : 'x') + (ok ? 'Correct!' : 'Not quite.') + '</strong> ' + q.explain;
      ex.classList.toggle('bad', !ok);
      ex.classList.add('show');
      $('#quizScore').innerHTML = icon('award') + 'Score: ' + score;
      $('#quizProgress').style.width = ((cur + 1) / qs.length * 100) + '%';
      $('#nextQ').disabled = false;
      $('#nextQ').focus({ preventScroll: true });
    }

    function renderResult() {
      var pct = Math.round(score / qs.length * 100);
      var prevBest = store('cybersafe_best');
      var best = Math.max(pct, prevBest || 0);
      store('cybersafe_best', best);

      var rank, msg;
      if (pct >= 90) { rank = 'Cyber Guardian'; msg = 'Outstanding! You have excellent cybersecurity awareness. Share what you know to help others stay safe.'; }
      else if (pct >= 70) { rank = 'Security Savvy'; msg = 'Great job! You know the essentials. Review the tips below to close the remaining gaps.'; }
      else if (pct >= 50) { rank = 'Getting There'; msg = 'Good start! A few habits need attention — check our Cybersecurity Tips section to level up.'; }
      else { rank = 'At Risk'; msg = 'Cyber criminals target people who are unaware. Read our tips and try the quiz again — you will improve quickly!'; }

      quizEl.innerHTML =
        '<div class="quiz-result">' +
          '<div class="score-ring">' +
            '<svg viewBox="0 0 160 160" aria-hidden="true">' +
              '<defs><linearGradient id="ringGrad" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="#2563eb"/><stop offset="1" stop-color="#38bdf8"/></linearGradient></defs>' +
              '<circle cx="80" cy="80" r="70" fill="none" stroke="#e8eef8" stroke-width="12"/>' +
              '<circle class="ring-val" id="ringVal" cx="80" cy="80" r="70" fill="none" stroke="url(#ringGrad)" stroke-width="12" stroke-linecap="round" stroke-dasharray="' + RING + '" stroke-dashoffset="' + RING + '"/>' +
            '</svg>' +
            '<div class="val">' + score + '/' + qs.length + '</div>' +
          '</div>' +
          '<span class="rank">' + rank + '</span>' +
          '<h3>You scored ' + pct + '%</h3>' +
          '<p>' + msg + '</p>' +
          (pct > (prevBest || 0) && prevBest !== null ? '<p class="best">New personal best! 🎉</p>' : '<p class="best">Best score: ' + best + '%</p>') +
          '<div class="result-actions">' +
            '<button class="btn btn-primary" id="retryQuiz">' + icon('refresh') + 'Try again</button>' +
            '<button class="btn btn-ghost" id="reviewBtn">' + icon('eye') + 'Review answers</button>' +
            '<a class="btn btn-ghost" href="#tips">' + icon('bulb') + 'Read tips</a>' +
          '</div>' +
          '<div class="review" id="review">' + answers.map(function (a, i) {
            return '<div class="review-item' + (a.ok ? '' : ' bad') + '"><b>' + (i + 1) + '. ' + a.q + '</b>' +
              '<span>Your answer: ' + a.chosen + '</span>' +
              (a.ok ? '' : '<span>Correct answer: ' + a.correct + '</span>') + '</div>';
          }).join('') + '</div>' +
        '</div>';

      setTimeout(function () {
        var ring = $('#ringVal');
        if (ring) ring.style.strokeDashoffset = RING * (1 - pct / 100);
      }, 60);
      $('#retryQuiz').addEventListener('click', startQuiz);
      $('#reviewBtn').addEventListener('click', function () {
        var r = $('#review'), open = r.classList.toggle('show');
        this.innerHTML = icon(open ? 'eye-off' : 'eye') + (open ? 'Hide answers' : 'Review answers');
      });
    }
    renderIntro();

    /* =========================================================
       CONTACT FORM
       ========================================================= */
    var form = $('#contactForm');
    var RULES = {
      name: function (v) { return v.trim().length >= 2 || 'Please enter your name (at least 2 characters).'; },
      email: function (v) { return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v.trim()) || 'Please enter a valid email address.'; },
      phone: function (v) { return v.trim() === '' || /^[+\d][\d\s\-()]{6,17}$/.test(v.trim()) || 'Please enter a valid phone number.'; },
      subject: function (v) { return v !== '' || 'Please choose a topic.'; },
      message: function (v) { return v.trim().length >= 10 || 'Your message should be at least 10 characters.'; }
    };
    function validateField(name) {
      var el = form.elements[name], field = el.closest('.field'), res = RULES[name](el.value);
      if (res !== true) {
        field.classList.add('error');
        field.querySelector('.err').textContent = res;
        return false;
      }
      field.classList.remove('error');
      return true;
    }
    Object.keys(RULES).forEach(function (name) {
      var el = form.elements[name];
      el.addEventListener(el.tagName === 'SELECT' ? 'change' : 'input', function () {
        if (el.closest('.field').classList.contains('error')) validateField(name);
      });
      el.addEventListener('blur', function () { if (el.value) validateField(name); });
    });
    form.elements.message.addEventListener('input', function () {
      $('#msgCount').textContent = this.value.length + ' / 1000';
    });

    form.addEventListener('submit', function (e) {
      e.preventDefault();
      var valid = Object.keys(RULES).map(validateField).every(Boolean);
      if (!valid) {
        var firstErr = form.querySelector('.field.error input, .field.error select, .field.error textarea');
        if (firstErr) firstErr.focus();
        toast('Please fix the highlighted fields.');
        return;
      }
      if (!form.elements.consent.checked) {
        toast('Please tick the consent box to continue.');
        form.elements.consent.focus();
        return;
      }
      var entry = {
        name: form.elements.name.value.trim(),
        email: form.elements.email.value.trim(),
        phone: form.elements.phone.value.trim(),
        subject: form.elements.subject.value,
        message: form.elements.message.value.trim(),
        date: new Date().toISOString()
      };
      var saved = store('cybersafe_messages') || [];
      saved.push(entry);
      store('cybersafe_messages', saved);

      $('#successText').textContent = 'Thank you, ' + entry.name.split(' ')[0] + '! Your message about “' + entry.subject + '” has been received. We will reply to ' + entry.email + ' soon.';
      $('#formCard').classList.add('sent');
      form.reset();
      $('#msgCount').textContent = '0 / 1000';
    });
    $('#sendAnother').addEventListener('click', function () {
      $('#formCard').classList.remove('sent');
      form.elements.name.focus();
    });

    /* =========================================================
       NEWSLETTER
       ========================================================= */
    $('#newsForm').addEventListener('submit', function (e) {
      e.preventDefault();
      var input = $('#newsEmail'), v = input.value.trim();
      if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) { toast('Please enter a valid email address.'); input.focus(); return; }
      var subs = store('cybersafe_subscribers') || [];
      if (subs.indexOf(v) === -1) subs.push(v);
      store('cybersafe_subscribers', subs);
      input.value = '';
      toast('Subscribed! Watch your inbox for security tips.');
    });
  })();
